prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
prefix foaf: <http://xmlns.com/foaf/0.1/> 
prefix dbo: <http://dbpedia.org/ontology/>
prefix dbp: <http://dbpedia.org/property/>
prefix dbr: <http://dbpedia.org/resource/>
prefix rdfs: <http://www.w3.org/2000/01/rdf-schema#>


Select Distinct ?book ?name ?publishName ?genre
Where {
       ?book rdf:type dbo:Book.
       ?book foaf:name ?name.
       ?book dbo:publisher ?publisher.
       ?publisher foaf:name ?publishName.
       ?book dbo:literaryGenre ?genre.
       Values ?genre {dbr:Fiction dbr:Science_fiction dbr:Horror_fiction dbr:Detective_fiction}.
       Filter regex(?publishName, "^Harper").
}
Limit 50




Select ?book ?name ?genre ?namePubl 
Where {
       ?book rdf:type dbo:Book.
       ?book foaf:name ?name.
       ?book dbo:publisher ?publisher.
       ?publisher foaf:name ?namePubl.
       ?book dbo:literaryGenre ?genre.
       Values ?genre {dbr:Science_fiction dbr:Horror_fiction dbr:Detective_fiction dbr:Historical_fiction}.
       Filter regex(?namePubl, "^Harper").
}
